# ModSync server settings. Only read on the machine running the server.
# Everything here is optional. Delete the # to turn a line on.
#
# ModSync already picks sensible values from the size of your Mods folder:
#
#   under 512 MB   12 downloaders at once, 20 MB/s shared
#   512 MB - 4 GB   6 downloaders at once, 15 MB/s shared
#   over 4 GB       3 downloaders at once, 10 MB/s shared
#
# It also gives less bandwidth to downloads while people are playing, so a new
# arrival pulling a big pack cannot lag out the players already in the world.
# Only set these if your host needs something different.


# Total upload used for mod downloads, shared by everyone downloading at once.
# Set this below your host's upload limit.
#MAXMBPERSEC=10

# How many players are served at the same time. The rest wait in line and are
# told their place. Fewer at once means each one finishes sooner.
#MAXDOWNLOADERS=3

# Mod folders clients should never receive. One line each, folder name only.
# Use this for server-only mods, admin tools, or anything with private data.
#EXCLUDE=SomeServerOnlyMod
#EXCLUDE=AnotherAdminTool

# TFP_CommandExtensions and Xample_MarkersMod ship with every dedicated server.
# ModSync never sends those two. An EXCLUDE line is not required for them.
