# Fill the lines below, then double-click GENERATE.bat.
# Zip the one file in Output\ and send that to players.
# Do not commit a filled SHARE_URL to a public repo.

SHARE_URL=
CLIENT_BAT_NAME=ClientSideModsSync.SERVERNAME
REMOVE_NOT_IN_CLOUD=1
DISCORD_WEBHOOK_URL=
DRY_RUN=0


# SHARE_URL
# Required. Players never log in.
# Paste a link anyone with the link can view.
# Use a personal OneDrive folder, a Google Drive folder, or a .zip download
# (Dropbox, Mega file, Google Drive file, or any https .zip).
# Mega folders and work/school SharePoint folders cannot be listed. Zip those.

# CLIENT_BAT_NAME
# Required. This is the .bat players double-click.
# Replace SERVERNAME with your server name.
# Example: ClientSideModsSync.MandaOutback

# REMOVE_NOT_IN_CLOUD
# 1 = player's Mods folder matches the pack. Extra local mods are deleted.
# 0 = only add/update pack mods. Extra local mods are left alone.

# DISCORD_WEBHOOK_URL
# Optional. Private webhook for install reports. Leave blank if unused.

# DRY_RUN
# 1 = generated bat shows the plan only and changes nothing (for testing).
# 0 = normal sync.
